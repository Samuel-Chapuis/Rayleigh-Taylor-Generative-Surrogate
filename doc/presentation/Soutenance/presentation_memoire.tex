        \small

\begin{frame}
  \frametitle{Rayleigh--Taylor instability}
  \framesubtitle{A nonlinear instability with a hierarchical spatial structure}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.46\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.68\textheight,keepaspectratio]{\reportfig{ch1/RTI_Evolution_f.png}}
      \par\vspace{0.5ex}
      \textit{Physical mechanism}
      \\[-0.2ex]
      % \smallref{Dense fluid accelerated by a lighter fluid: bubbles and spikes develop.}
    \end{column}
    \begin{column}{0.50\textwidth}
      \begin{block}{Characteristic stages}
        \small
        \begin{itemize}
          \item growth of initial interface perturbations;
          \item formation of bubbles and spikes;
          \item sharp gradients and localized structures;
          \item mixing and transfer of energy towards finer scales.
        \end{itemize}
      \end{block}
      \vspace{1ex}
      % \centering
      \begin{alertblock}{Objective}
        \small A surrogate model must capture both global morphology and local fluctuations.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[plain]
  \frametitle{Why wavelets for RTI fields?}
  \framesubtitle{Localize information in both position and spatial scale}
  \vspace{10pt}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.68\textheight,keepaspectratio]{\reportfig{ch2/wavelet_decomposition.png}}
      \par\vspace{0.5ex}
      \textit{Two-dimensional wavelet decomposition}
    \end{column}
    \begin{column}{0.39\textwidth}
      \begin{block}{Limitation of a global representation}
        \small Fourier modes are useful for spectral analysis, but they are spatially global and less adapted to localized sharp interfaces.
      \end{block}
      \begin{block}{Wavelet inductive bias}
        \small A multi-resolution representation separates large-scale organization from localized directional details.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.5ex}
  \begin{alertblock}{Why this matters for WSGM}
    \centering\small The coarse field carries global organization; the detail channels carry localized fine-scale variability.
  \end{alertblock}
\end{frame}

\begin{frame}[plain]
  \frametitle{Wavelet representation of RTI fields}
  \framesubtitle{One approximation channel and three directional detail channels}
  \vspace{10pt}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.8\textheight,keepaspectratio]{\reportfig{ch2/2DExemple_Masque.png}}
      \par\vspace{0.5ex}
      \textit{Coefficient organization}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{At wavelet level $j$}
        \small
        \begin{itemize}
          \item $LL$: low-frequency approximation;
          \item $LH$, $HL$, $HH$: horizontal, vertical and diagonal details;
          \item each sub-band has a reduced spatial resolution.
        \end{itemize}
      \end{block}
      \begin{alertblock}{Interpretation and intuition}
        \begin{itemize}
          \item The $LL$ sub-band captures the overall shape and large-scale features.
          \item The details can be generated conditionally on the coarse field.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{0.2ex}
\end{frame}

% 4
\begin{frame}
  \frametitle{Why a generative surrogate?}
  \framesubtitle{The target is a distribution of fields, not one reconstructed trajectory}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{block}{Learning objective}
        \small Approximate the distribution represented by the DNS ensemble, then sample new fields from the learned model.
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{alertblock}{Two strategies compared}
        \small
        \begin{enumerate}
          \item direct score-based diffusion in image space;
          \item wavelet score-based diffusion with a coarse-to-fine cascade.
        \end{enumerate}
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{0.7ex}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{block}{Physical requirements}
        \small Preserve RTI morphology, the mixing-layer organization, density statistics and scale-dependent fluctuations.
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{exampleblock}{Central hypothesis}
        \small Separating coarse organization from localized details may reduce the cost of generation, provided that the coarse model converges to the correct distribution.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% 6
\begin{frame}[plain]
  \frametitle{Diffusion: noising then denoising}
  \framesubtitle{From a data distribution to Gaussian noise and back}
  \vspace{15pt}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.57\textheight,keepaspectratio]{\reportfig{ch3/diffusion_1d_forward.png}}
      \par\vspace{0.5ex}
      % \textbf{Forward diffusion: destroy structure}
      % \\[-0.2ex]
      % \smallref{The data distribution progressively approaches a Gaussian prior.}
      \begin{block}{Forward process : destroy structure}
        \footnotesize Known fields are progressively corrupted until their structure is replaced by noise.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.57\textheight,keepaspectratio]{\reportfig{ch3/diffusion_1d_reverse_score.png}}
      \par\vspace{0.5ex}
      % \textbf{Reverse diffusion: recover samples}
      % \\[-0.2ex]
      % \smallref{The learned score guides trajectories back towards the data distribution.}
      \begin{block}{Reverse process : recover samples}
        \footnotesize The learned denoising direction is applied repeatedly to recover a plausible sample.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.5ex}
  \begin{alertblock}{Core idea}
    \centering\small Train on noisy fields, then generate by reversing the same path from Gaussian noise.
  \end{alertblock}
\end{frame}

% 7
\begin{frame}
  \frametitle{How score-based diffusion works}
  \framesubtitle{The network learns a denoising direction at every noise level}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \vspace{20pt}
      \centering
      \includegraphics[width=\linewidth,height=0.65\textheight,keepaspectratio]{\reportfig{ch4/training_loop2.drawio.png}}
      \par\vspace{0.5ex}
    \end{column}
    \begin{column}{0.39\textwidth}
      \begin{block}{During training}
        \small
        \begin{enumerate}
          \item start from a known physical field;
          \item add noise at a randomly selected level;
          \item train the network to predict the direction back towards realistic data.
        \end{enumerate}
      \end{block}
      \begin{alertblock}{Generation}
        \footnotesize Start from noise and follow the learned direction repeatedly. Fewer steps are faster but less accurate.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

        \small

\begin{frame}
  \frametitle{Direct image-space SGM}
  \framesubtitle{Time-conditioned U-Net for full-field score prediction}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.54\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.60\textheight,keepaspectratio]{\reportfig{ch4/Unet2.drawio.png}}
      \par\vspace{0.5ex}
      \textit{Baseline score network}
      \\[-0.2ex]
      \smallref{Convolutional U-Net with skip connections and time conditioning.}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{Training}
        \small
        \begin{itemize}
          \item continuous VP-SDE;
          \item random diffusion time for each batch;
          \item $v$-parameterization and mean-squared error loss;
          \item sinusoidal time embedding injected at each scale.
        \end{itemize}
      \end{block}
      \begin{block}{Sampling}
        \small Start from Gaussian noise and integrate the probability-flow ODE with a Heun solver.
      \end{block}
      \vspace{-0.4cm}
      \smallref{U-Net: Ronneberger et al.; implementation: PyTorch.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}
  \frametitle{Conditional diffusion for wavelet details}
  \framesubtitle{The score model operates only on fine-scale coefficients}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.65\textheight,keepaspectratio]{\reportfig{ch4/training_loop_conditioned.drawio.png}}
      \par\vspace{0.5ex}
      \textit{Conditional training loop}
      \\[-0.2ex]
      \smallref{Noise is added to details; $LL$ remains the conditioning field.}
    \end{column}
    \begin{column}{0.39\textwidth}
      \begin{block}{Model input}
        \small $[LL,D_t]$ with $D_t=(LH_t,HL_t,HH_t)$ noisy.
      \end{block}
      \begin{block}{Model output}
        \small Prediction of the score-model target associated with the detail tensor.
      \end{block}
      \begin{alertblock}{Key condition}
        \small The detail model can only generate compatible fine-scale content if the coarse field is physically and statistically consistent.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}
  \frametitle{Experimental protocol and metrics}
  \framesubtitle{Compare detail quality with coarse-model quality}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.45\textwidth}
      \begin{block}{Reference-coarse WSGM}
        \small
        \begin{itemize}
          \item $LL$ extracted from a validation field;
          \item details generated conditionally;
          \item oracle ablation, not an autonomous generator.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{alertblock}{Generated-coarse WSGM}
        \small
        \begin{itemize}
          \item $LL$ sampled by a coarse SGM;
          \item fully generative cascade;
          \item measures the end-to-end result.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{1.2ex}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.45\textwidth}
      \begin{block}{Distributional metric}
        \small PhyFID compares ensembles in a feature space learned from RTI fields.
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Physical diagnostics}
        \small Mean vertical density profiles and fluctuation-energy spectra complement the feature-space distance.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}
  \frametitle{Central result: a quality--speed trade-off}
  \framesubtitle{The WSGM is faster, but currently less accurate in its fully generative form}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.43\textwidth}
      \begin{block}{Direct SGM}
        \small
        \begin{itemize}
          \item best overall sample quality;
          \item generation in the full image space;
          \item higher sampling cost.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{alertblock}{Fully generative WSGM}
        \small
        \begin{itemize}
          \item faster generation through multi-scale factorization;
          \item rougher fields and larger distance to validation data;
          \item speed advantage depends on coarse-model convergence.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{1.8ex}
  \centering
  {\large\color{CEAred}\textbf{WSGM does not outperform SGM yet: it provides a credible route towards faster sampling.}}
\end{frame}

\begin{frame}
  \frametitle{Limitations, perspectives and conclusion}
  \framesubtitle{Improve the coarse model before exploiting the surrogate}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.46\textwidth}
      \begin{alertblock}{Current limitations}
        \small
        \begin{itemize}
          \item limited DNS ensemble and partial parameter coverage;
          \item wavelet architecture remains exploratory;
          \item fully generated WSGM fields are still rougher;
          \item the reference-coarse setting is diagnostic, not autonomous.
        \end{itemize}
      \end{alertblock}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Next steps}
        \small
        \begin{itemize}
          \item improve and evaluate the coarse diffusion model;
          \item perform a systematic hyperparameter analysis;
          \item target rare-event prediction with fast ensembles;
          \item address inverse problems: infer initial conditions compatible with an observed morphology.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{-0.45cm}
  \begin{exampleblock}{Take-home message}
    \centering\footnotesize The direct SGM gives the best quality. The WSGM is faster, but its competitiveness depends on convergence of the generated coarse state.
  \end{exampleblock}
\end{frame}
